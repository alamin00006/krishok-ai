import { Alert } from 'react-native';
import { IconButton } from '../../components/ui';
import type { Theme } from '../../theme';

export function VoiceInputButton({
  theme,
  onTranscript,
}: {
  theme: Theme;
  onTranscript: (text: string) => void;
}) {
  const start = async () => {
    try {
      const { ExpoSpeechRecognitionModule } = await import('expo-speech-recognition');
      const subscriptions = [
        ExpoSpeechRecognitionModule.addListener('result', (event) => {
          const transcript = event.results[0]?.transcript?.trim();
          if (transcript) onTranscript(transcript);
        }),
        ExpoSpeechRecognitionModule.addListener('error', (event) => {
          Alert.alert('ভয়েস ইনপুট', event.message || 'কথা শনাক্ত করা যায়নি।');
          subscriptions.forEach((subscription) => subscription.remove());
        }),
        ExpoSpeechRecognitionModule.addListener('end', () => {
          subscriptions.forEach((subscription) => subscription.remove());
        }),
      ];
      const permission = await ExpoSpeechRecognitionModule.requestPermissionsAsync();
      if (!permission.granted) {
        subscriptions.forEach((subscription) => subscription.remove());
        Alert.alert('অনুমতি প্রয়োজন', 'ভয়েস ইনপুটের জন্য মাইক্রোফোন অনুমতি দিন।');
        return;
      }
      ExpoSpeechRecognitionModule.start({
        lang: 'bn-BD',
        interimResults: true,
        continuous: false,
      });
    } catch {
      Alert.alert(
        'Development build প্রয়োজন',
        'Voice input Expo Go-তে চলে না। npx expo run:android দিয়ে development build চালান।'
      );
    }
  };

  return <IconButton theme={theme} icon="mic-outline" onPress={() => void start()} />;
}
